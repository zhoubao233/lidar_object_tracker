cmake_minimum_required(VERSION 3.10)
project(lidar_object_tracker)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(catkin REQUIRED COMPONENTS roscpp roslib sensor_msgs nav_msgs std_msgs geometry_msgs visualization_msgs livox_ros_driver2 rosbag gazebo_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common kdtree)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED jsoncpp)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES tracker_core tracker_ros tracker_livox
 CATKIN_DEPENDS roscpp roslib sensor_msgs nav_msgs std_msgs geometry_msgs visualization_msgs livox_ros_driver2 rosbag gazebo_msgs
 DEPENDS EIGEN3 PCL)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${JSONCPP_INCLUDE_DIRS})
add_library(tracker_core src/tracking_core.cpp)
target_link_libraries(tracker_core ${PCL_LIBRARIES} ${JSONCPP_LIBRARIES})
add_library(tracker_ros src/ros_utils.cpp)
target_link_libraries(tracker_ros tracker_core ${catkin_LIBRARIES} yaml-cpp)
add_library(tracker_livox src/livox_core.cpp src/livox_adapter.cpp)
add_dependencies(tracker_livox ${catkin_EXPORTED_TARGETS})
target_link_libraries(tracker_livox tracker_ros)
add_executable(object_tracker src/object_tracker.cpp)
target_link_libraries(object_tracker tracker_ros Threads::Threads)
foreach(node livox_to_body livox_real_to_body)
 add_executable(${node} src/${node}.cpp)
 target_link_libraries(${node} tracker_livox)
 add_dependencies(${node} ${catkin_EXPORTED_TARGETS})
endforeach()
install(TARGETS tracker_core tracker_ros tracker_livox object_tracker livox_to_body livox_real_to_body
 ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY launch config rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES LICENSE NOTICE.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY licenses DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

if(CATKIN_ENABLE_TESTING)
 find_package(rostest REQUIRED)
 catkin_add_gtest(test_tracker tests/test_tracking.cpp tests/test_livox.cpp)
 if(TARGET test_tracker)
  target_link_libraries(test_tracker tracker_livox)
  add_dependencies(test_tracker ${catkin_EXPORTED_TARGETS})
 endif()
 add_executable(pipeline_check tests/check_pipeline_ros.cpp)
 target_link_libraries(pipeline_check tracker_livox gtest)
 add_dependencies(pipeline_check ${catkin_EXPORTED_TARGETS})
 add_rostest(tests/simulation.test DEPENDENCIES pipeline_check object_tracker livox_to_body)
 add_rostest(tests/real.test DEPENDENCIES pipeline_check object_tracker livox_real_to_body)
endif()

add_executable(evaluate_bag tools/evaluate_bag.cpp)
add_dependencies(evaluate_bag ${catkin_EXPORTED_TARGETS})
target_link_libraries(evaluate_bag tracker_ros)
install(TARGETS evaluate_bag RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
